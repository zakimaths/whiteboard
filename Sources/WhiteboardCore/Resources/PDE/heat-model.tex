\partial_t u &= \kappa\,\partial_{xx}u,
&&0<x<L,\quad t>0,\\[4pt]
u(0,t)&=u(L,t)=0, &&\kappa>0,\\[4pt]
u(x,0)&=U_0\!\left[\sin\!\frac{\pi x}{L}
       +0.35\sin\!\frac{3\pi x}{L}\right],\\[7pt]
\xi&=x/L,\qquad \tau=\kappa t/L^2,
&&v=u/U_0,\\
v_\tau&=v_{\xi\xi}, &&0<\xi<1.
\end{aligned}
