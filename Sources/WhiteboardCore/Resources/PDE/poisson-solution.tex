\begin{aligned}
u(x,y)&=\sin(\pi x)\sin(\pi y),\\[5pt]
u_{xx}&=-\pi^2u,\qquad u_{yy}=-\pi^2u,\\[4pt]
-\Delta u&=2\pi^2u=f,\qquad u|_{\partial\Omega}=0,\\[6pt]
u(\tfrac12,\tfrac12)&=1,\qquad \max_{\bar\Omega}u=1,\\[5pt]
\int_\Omega |\nabla u|^2\,d\Omega
 &=\int_\Omega fu\,d\Omega=\frac{\pi^2}{2}.
\end{aligned}
