\begin{aligned}
r&=c\Delta t/\Delta x,\qquad 0<r\le1,\\[5pt]

 +r^2(u_{j-1}^n-2u_j^n+u_{j+1}^n),\\[5pt]
&j=1,\ldots,N-1,\quad n\ge1,\\[3pt]
u_j^1&=u_j^0+\Delta t\,g_j
