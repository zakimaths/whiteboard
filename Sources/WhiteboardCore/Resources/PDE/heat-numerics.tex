x_j&=j\Delta x,\quad t_n=n\Delta t,
&r&=\frac{\kappa\Delta t}{(\Delta x)^2},\\[5pt]

&j=1,\ldots,N-1,\quad n\ge0,\\[3pt]

&&\text{(explicit stability)},\\[5pt]
