\begin{aligned}
-\Delta u&=f &&\text{in }\Omega=(0,1)^2,\\[5pt]
u&=0 &&\text{on }\partial\Omega,\\[5pt]
f(x,y)&=2\pi^2\sin(\pi x)\sin(\pi y),\\[7pt]
\text{Find }u&\in H_0^1(\Omega):\\[-2pt]
\int_\Omega\nabla u\cdot\nabla v\,d\Omega
 &=\int_\Omega fv\,d\Omega
 &&\forall v\in H_0^1(\Omega).
\end{aligned}
